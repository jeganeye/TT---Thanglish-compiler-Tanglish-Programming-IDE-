Conditional Statements
mathippu = 15

enna mathippu > 10:
    eluthu('எண் 10ஐ விட பெரியது')
illainu mathippu == 10:
    eluthu('எண் 10க்கு சமம்')
illana:
    eluthu('எண் 10ஐ விட சிறியது')
Loops
# For loop
eluthu('For Loop:')
payan i udaya range(5):
    eluthu(i)

# While loop
eluthu('\nWhile Loop:')
count = 0
neram count < 5:
    eluthu(count)
    count = count + 1
Functions
varu vanakkam(peyar):
    mudivu f'வணக்கம் {peyar}!'

message = vanakkam('தமிழ்')
eluthu(message)

# Function with multiple parameters
varu koottu(a, b):
    mudivu a + b

eluthu('5 + 10 =', koottu(5, 10))
Data Structures
# List (pattial)
fruits = ['ஆப்பிள்', 'ஆரஞ்சு', 'வாழைப்பழம்']
eluthu('பழங்கள்:', fruits)
fruits.serkkavum('திராட்சை')
eluthu('புதிய பழம் சேர்த்த பின்:', fruits)

# Dictionary (agarathi)
person = {
    'peyar': 'ராஜா',
    'vayathu': 30,
    'oor': 'சென்னை'
}
eluthu('நபர்:', person)
eluthu('பெயர்:', person['peyar'])
User Input
# Get user input
name = vaangu('Enter your name: ')
eluthu('வணக்கம்', name, '!')

# Calculate age
birth_year = enn(vaangu('Enter your birth year: '))
current_year = 2025
age = current_year - birth_year
eluthu('Your age is:', age)
